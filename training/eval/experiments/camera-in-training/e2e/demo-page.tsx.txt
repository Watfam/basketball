"use client";
// Temporary demo route used by the two e2e scripts here. To rerun them: save this as src/app/lab/p10-demo/page.tsx, allow "/lab/p10-demo" in

import { useSearchParams } from "next/navigation";
import { Suspense } from "react";
import { SessionPlayer, type SessionDrill } from "@/components/session-player";
import { ShootingHub } from "@/components/shooting-hub";
import type { DrillStart } from "@/lib/basketball/drill-start";
import { BENCHMARKS } from "@/lib/basketball/benchmarks";

const P = "11111111-1111-4111-8111-111111111111";
const W = "22222222-2222-4222-8222-222222222222";
const E1 = "33333333-3333-4333-8333-333333333331";
const E2 = "33333333-3333-4333-8333-333333333332";
const E3 = "33333333-3333-4333-8333-333333333333";

const mk = (id: string, name: string, sort: number, reps: number | null): SessionDrill => ({
  id, drill_id: id.replace(/3/g, "4"), sort_order: sort, block: "Main", variant_label: name === "Free Throw Reset" ? "Right wing" : null, levels: null, level_targets: null,
  target_sets: 2, target_reps: reps, target_duration_seconds: null,
  drills: { id: id.replace(/3/g, "4"), name, description: "Demo drill", video_url: null, source_trainer: null, setup: null, cues: ["Keep your elbow under the ball"], common_mistakes: null, equipment: null, film_resources: [] },
});
const drills = [mk(E1, "Free Throw Reset", 1, 10), mk(E2, "Form Shooting Progression", 2, 10), mk(E3, "Ball Handling Warmup", 3, 10)];
const lesson = { id: "l1", title: "Building a Repeatable Release", watchFor: ["Same dip every time"], url: "https://www.youtube.com/watch?v=abc", basis: "linked" as const, watched: false, more: [], studySession: null };
const extras = {
  [E1]: { lesson, tracking: { trackable: true, category: "stationary" as const, categoryLabel: "Stationary shooting", why: "A free throw is shot standing still.", goal: { kind: "shots" as const, target: 20 } }, benchmark: null },
  [E2]: { lesson: null, tracking: { trackable: false, category: "stationary" as const, categoryLabel: "Stationary shooting", why: "Form work is about how you shoot, not whether it goes in.", goal: null }, benchmark: null },
};
const tracked = { type: "shooting" as const, variant: "Right wing", category: "stationary" as const, category_basis: "drill_definition" as const, makes: 14, attempts: 20, target: { kind: "shots" as const, value: 20 }, tracking: { v: 1 as const, source: "camera" as const, shot_session_id: "44444444-4444-4444-8444-444444444444", camera_attempts: 19, camera_makes: 13, flipped: 1, added_by_hand: 1, rule_version: "v4", model_version: "ball6", adjusted: null } };

function Demo() {
  const v = useSearchParams().get("view") ?? "workout";
  const ctx = { workoutSessionId: W, workoutDrillId: E1, drillName: "Free Throw Reset", variant: "Right wing", goal: { kind: "shots" as const, target: 5 } };
  const start: DrillStart = { context: ctx, categoryLabel: "Stationary shooting", planLabel: "Week 2 · Day 1", workoutName: "Shooting Day", lesson, benchmark: null };
  if (v === "bm-cs" || v === "bm-clock" || v === "bm-ft") {
    const bm = BENCHMARKS.find((b) => b.key === (v === "bm-cs" ? "catch_and_shoot" : v === "bm-clock" ? "beat_the_clock" : "free_throw"))!;
    // the clock benchmark runs 10 seconds here (the shortest the app allows) so the whole flow can be driven in a test
    const goal = bm.clockSeconds ? { kind: "time" as const, target: 10 } : { kind: "shots" as const, target: 5 };
    const c2 = { workoutSessionId: W, workoutDrillId: E1, drillName: bm.drillName, variant: bm.variantLabel, goal };
    return <div className="mx-auto w-full max-w-md p-4"><ShootingHub playerId={P} playerName="Demo" suggestedLabels={[]} drill={{ context: c2, categoryLabel: "Benchmark", planLabel: "Week 3 · Day 4", workoutName: "Benchmark Check", lesson: lesson, benchmark: bm }}><p id="history">history</p></ShootingHub></div>;
  }
  if (v === "bm-workout") {
    const bm = BENCHMARKS[0];
    const entry = { ...mk(E1, bm.drillName, 1, 10), variant_label: bm.variantLabel, target_sets: 3, target_reps: 10 };
    const ex = { [E1]: { lesson: { ...lesson, more: [{ id: "l2", title: "Curry: Feet Ready Before the Ball Arrives", url: "https://www.youtube.com/watch?v=def", watched: false }], studySession: { id: "s1", name: "Get Open Without the Ball", lessonCount: 3 } }, tracking: { trackable: true, category: "stationary" as const, categoryLabel: "Stationary shooting", why: "x", goal: { kind: "shots" as const, target: 30 } }, benchmark: bm } };
    return <SessionPlayer playerId={P} sessionId={W} workoutName="Benchmark Check" drills={[entry]} level="intermediate" volumeStep={2} alreadyCompleted={false} initialLoggedEntryIds={[]} entryExtras={ex} />;
  }
  if (v === "workout" || v === "result" || v === "waiting")
    return <SessionPlayer playerId={P} sessionId={W} workoutName="Shooting Day" drills={drills} level="intermediate" volumeStep={0} alreadyCompleted={false} initialLoggedEntryIds={v === "waiting" ? [E3] : []} entryExtras={extras} trackedResults={v === "result" ? { [E1]: tracked } : {}} focusEntryId={v === "result" ? E1 : null} />;
  return (
    <div className="mx-auto w-full max-w-md p-4">
      <ShootingHub playerId={P} playerName="Demo" suggestedLabels={[]} drill={v === "problem" ? { problem: "This drill is about form, so there is nothing for the camera to count." } : start}>
        <p id="history">history</p>
      </ShootingHub>
    </div>
  );
}
export default function Page() { return <Suspense><Demo /></Suspense>; }
